// Footstep boundaries and furniture collision. Movement sweeps on each axis to glide at corners.
function collisionBounds(x,z){
  let half=z<7?3.42:z<11.15?2.57:3.42;
  if(z<.25||z>16.14)return false;
  if(Math.abs(x)>half)return false;
  if((z>6.87&&z<7.13)&&Math.abs(x)>1.20)return false;
  if((z>11.02&&z<11.28)&&Math.abs(x)>1.08)return false;
  if(z>=7&&z<=11.15&&Math.abs(x)>2.57)return false;
  const r=.27;
  for(const o of solids)if(Math.abs(x-o.x)<o.w/2+r&&Math.abs(z-o.z)<o.d/2+r)return false;
  return true;
}

    let side=(keys.has('d')||keys.has('arrowright')?1:0)-(keys.has('a')||keys.has('arrowleft')?1:0);
    const mag=Math.hypot(f,side)||1;f/=mag;side/=mag;
    const speed=(keys.has('shift')?1.35:2.65)*dt;
    const dx=(Math.sin(yaw)*f+Math.cos(yaw)*side)*speed;
    const dz=(Math.cos(yaw)*f-Math.sin(yaw)*side)*speed;

  }
