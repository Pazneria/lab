function canStand(x,z){
  const outside=z>7.95;
  if(outside){if(Math.abs(x)>4.68||z>10.08)return false;}
  else if(z< -7.48||x < -4.19)return false;
  if(!outside&&x>4.19 && !(z>-7.48&&z<-4.58))return false;
  if(x>7.82)return false;
  if(x>-1.76&&x<1.76&&z>-6.46&&z<5.56)return false;
  // keep a soft shoulder from the workbench and the fixed jars
  if(x>4.84&&x<8.18&&z>-7.82&&z<-6.2)return false;
  return true;
}
function move(dt){
  const forward=(keys.has('KeyW')||keys.has('ArrowUp')?1:0)-(keys.has('KeyS')||keys.has('ArrowDown')?1:0);
  const strafe=(keys.has('KeyD')||keys.has('ArrowRight')?1:0)-(keys.has('KeyA')||keys.has('ArrowLeft')?1:0);
  if(!forward&&!strafe)return;
  const norm=1/Math.hypot(forward,strafe),pace=keys.has('ShiftLeft')||keys.has('ShiftRight')?2.3:3.05;
  const f=forward*norm*pace*dt,s=strafe*norm*pace*dt,yaw=player.yaw;
  const dx=-Math.sin(yaw)*f+Math.cos(yaw)*s,dz=-Math.cos(yaw)*f-Math.sin(yaw)*s;
  if(canStand(player.x+dx,player.z))player.x+=dx;
  if(canStand(player.x,player.z+dz))player.z+=dz;
  camera.position.set(player.x,1.64,player.z);
}
